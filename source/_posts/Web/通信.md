### BroadCast Channel

```js
const broadcastChannel = new BroadcastChannel('channel-id');

broadcastChannel.addEventListener('message', (e) => {
	console.log(e.data);
})

broadcastChannel.postMessage('post message');

broadcastChannel.close();
```

### Service Worker

*serviceWorker.js*

```js
this.addEventListener('message', function (event) {
  console.log('worker message')
  this.clients.matchAll().then(function (clients) {
    clients.forEach(function (client) {
      // 这里的判断目的是过滤掉当前 Tab 页面，也可以使用 visibilityState 的状态来判断
      if (!client.focused) {
        client.postMessage(event.data)
      }
    })
  })
})
```

*main.js*

```js
navigator.serviceWorker
	.register('serviceWorker.js')
	.then(function (registration) {
		console.log('ServiceWorker registration successful with scope: ', registration.scope)
	})
	.catch(function (err) {
		console.log('ServiceWorker registration failed: ', err)
	})

navigator.serviceWorker.addEventListener('message', function (e) {
	const data = e.data
	console.log('page message:', data)
	message.value.push(data)
})

navigator.serviceWorker.controller?.postMessage(Date.now().toString())
```

*main.js*

```js
const worker = new SharedWorker('sharedWorker.js')
worker.port.onmessage = (e) => {
	console.log('post', e.data)
	if (e.data.length > 0 && e.data.length !== message.value.length) {
		message.value.push(...e.data)
	}
}
worker.port.start()
```
